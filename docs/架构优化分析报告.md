```python
class ZeroCopyBatchProcessor:
    """零拷贝批处理器"""
    def __init__(self, batch_size=16):
        self.batch_size = batch_size
        self.image_buffer = np.empty((batch_size, 640, 640, 3), dtype=np.uint8)
        self.result_buffer = [None] * batch_size
    
    def process_batch_zero_copy(self, frame_buffers):
        """零拷贝批处理"""
        actual_size = min(len(frame_buffers), self.batch_size)
        
        # 零拷贝数据准备
        for i in range(actual_size):
            np.copyto(self.image_buffer[i], frame_buffers[i].get_frame_view())
        
        # 批处理分析
        results = self.analyzer.batch_analyze(self.image_buffer[:actual_size])
        
        # 结果映射
        for i, result in enumerate(results):
            self.result_buffer[i] = result
            
        return self.result_buffer[:actual_size]
```

```python
class AsyncAnalyzerLoader:
    """异步分析器加载器"""
    def __init__(self):
        self.loading_tasks = {}
        self.loaded_analyzers = {}
        self.load_lock = asyncio.Lock()
    
    async def load_analyzer_async(self, model_code):
        """真正的异步加载"""
        if model_code in self.loaded_analyzers:
            return self.loaded_analyzers[model_code]
        
        async with self.load_lock:
            if model_code in self.loading_tasks:
                return await self.loading_tasks[model_code]
            
            task = asyncio.create_task(self._load_analyzer_impl(model_code))
            self.loading_tasks[model_code] = task
            
            try:
                analyzer = await task
                self.loaded_analyzers[model_code] = analyzer
                return analyzer
            finally:
                del self.loading_tasks[model_code]
    
    async def _load_analyzer_impl(self, model_code):
        """分析器加载实现"""
        analyzer_class = AnalyzerRegistry.get_analyzer(model_code)
        analyzer = analyzer_class()
        await analyzer.load_model_async()
        return analyzer
```

```python
from typing import Dict, Set, Callable

class PluginMemoryManager:
    """插件内存管理器"""
    def __init__(self):
        self.weak_references: Dict[str, weakref.ref] = {}
        self.cleanup_callbacks: Dict[str, Set[Callable]] = {}
        self.module_cache: Dict[str, weakref.ref] = {}
    
    def register_plugin(self, plugin_name: str, plugin_instance):
        """使用弱引用注册插件"""
        def cleanup_callback(ref):
            self._on_plugin_cleanup(plugin_name)
        
        self.weak_references[plugin_name] = weakref.ref(plugin_instance, cleanup_callback)
        plugin_instance.add_cleanup_callback(self._on_plugin_cleanup)
    
    def register_module(self, module_name: str, module):
        """注册模块弱引用"""
        self.module_cache[module_name] = weakref.ref(module)
    
    def _on_plugin_cleanup(self, plugin_name: str):
        """插件清理回调"""
        if plugin_name in self.cleanup_callbacks:
            for callback in self.cleanup_callbacks[plugin_name]:
                try:
                    callback(plugin_name)
                except Exception as e:
                    logger.warning(f"插件清理回调失败: {e}")
            del self.cleanup_callbacks[plugin_name]
        
        # 清理弱引用
        if plugin_name in self.weak_references:
            del self.weak_references[plugin_name]
    
    def get_plugin(self, plugin_name: str):
        """获取插件实例"""
        weak_ref = self.weak_references.get(plugin_name)
        if weak_ref is not None:
            return weak_ref()
        return None
    
    def cleanup_all(self):
        """清理所有资源"""
        for plugin_name in list(self.weak_references.keys()):
            self._on_plugin_cleanup(plugin_name)
        
        self.weak_references.clear()
        self.cleanup_callbacks.clear()
        self.module_cache.clear()
```

#### 2.3.2 线程安全的状态管理

```python
import asyncio
from enum import Enum
from typing import Dict, Callable, Optional

class PluginStatus(Enum):
    UNLOADED = "unloaded"
    LOADING = "loading"
    LOADED = "loaded"
    INITIALIZING = "initializing"
    ACTIVE = "active"
    ERROR = "error"
    STOPPING = "stopping"
    STOPPED = "stopped"

class PluginStateMachine:
    """线程安全的插件状态机"""
    
    # 定义合法的状态转换
    VALID_TRANSITIONS = {
        PluginStatus.UNLOADED: [PluginStatus.LOADING],
        PluginStatus.LOADING: [PluginStatus.LOADED, PluginStatus.ERROR],
        PluginStatus.LOADED: [PluginStatus.INITIALIZING, PluginStatus.ERROR],
        PluginStatus.INITIALIZING: [PluginStatus.ACTIVE, PluginStatus.ERROR],
        PluginStatus.ACTIVE: [PluginStatus.STOPPING, PluginStatus.ERROR],
        PluginStatus.STOPPING: [PluginStatus.STOPPED, PluginStatus.ERROR],
        PluginStatus.STOPPED: [PluginStatus.LOADING, PluginStatus.UNLOADED],
        PluginStatus.ERROR: [PluginStatus.LOADING, PluginStatus.UNLOADED]
    }
    
    def __init__(self, plugin_name: str):
        self.plugin_name = plugin_name
        self.current_state = PluginStatus.UNLOADED
        self.state_lock = asyncio.Lock()
        self.state_callbacks: Dict[PluginStatus, Set[Callable]] = {}
        self.transition_history = []
    
    async def transition_to(self, new_state: PluginStatus) -> bool:
        """原子状态转换"""
        async with self.state_lock:
            if not self._is_valid_transition(self.current_state, new_state):
                logger.warning(f"插件 {self.plugin_name} 无效状态转换: {self.current_state} -> {new_state}")
                return False
            
            old_state = self.current_state
            self.current_state = new_state
            
            # 记录转换历史
            self.transition_history.append({
                'from': old_state,
                'to': new_state,
                'timestamp': asyncio.get_event_loop().time()
            })
            
            # 触发状态变更回调
            await self._trigger_state_callbacks(old_state, new_state)
            
            logger.info(f"插件 {self.plugin_name} 状态转换: {old_state} -> {new_state}")
            return True
    
    def _is_valid_transition(self, from_state: PluginStatus, to_state: PluginStatus) -> bool:
        """检查状态转换是否合法"""
        return to_state in self.VALID_TRANSITIONS.get(from_state, [])
    
    async def _trigger_state_callbacks(self, old_state: PluginStatus, new_state: PluginStatus):
        """触发状态变更回调"""
        callbacks = self.state_callbacks.get(new_state, set())
        for callback in callbacks:
            try:
                if asyncio.iscoroutinefunction(callback):
                    await callback(self.plugin_name, old_state, new_state)
                else:
                    callback(self.plugin_name, old_state, new_state)
            except Exception as e:
                logger.error(f"状态回调执行失败: {e}")
    
    def add_state_callback(self, state: PluginStatus, callback: Callable):
        """添加状态回调"""
        if state not in self.state_callbacks:
            self.state_callbacks[state] = set()
        self.state_callbacks[state].add(callback)
    
    def get_current_state(self) -> PluginStatus:
        """获取当前状态"""
        return self.current_state
    
    def get_transition_history(self) -> list:
        """获取状态转换历史"""
        return self.transition_history.copy()
```

#### 2.3.3 优化的依赖解析器

```python
from typing import Dict, List, Set, Optional
import hashlib
import json

class OptimizedDependencyResolver:
    """优化的依赖解析器"""
    
    def __init__(self):
        self.dependency_cache: Dict[str, List[str]] = {}
        self.resolution_cache: Dict[str, List[str]] = {}
        self.dependency_graph: Dict[str, Set[str]] = {}
        self.reverse_graph: Dict[str, Set[str]] = {}
        self.cache_version = 0
    
    def set_dependencies(self, plugin_name: str, dependencies: List[str]):
        """设置插件依赖"""
        if plugin_name not in self.dependency_graph:
            self.dependency_graph[plugin_name] = set()
        
        old_deps = self.dependency_graph[plugin_name].copy()
        new_deps = set(dependencies)
        
        # 更新正向图
        self.dependency_graph[plugin_name] = new_deps
        
        # 更新反向图
        for dep in old_deps - new_deps:
            if dep in self.reverse_graph:
                self.reverse_graph[dep].discard(plugin_name)
        
        for dep in new_deps - old_deps:
            if dep not in self.reverse_graph:
                self.reverse_graph[dep] = set()
            self.reverse_graph[dep].add(plugin_name)
        
        # 清除缓存
        self._invalidate_cache()
    
    def resolve_dependencies(self, plugin_names: List[str]) -> List[str]:
        """解析插件依赖顺序"""
        # 生成缓存键
        cache_key = self._generate_cache_key(plugin_names)
        
        # 检查缓存
        if cache_key in self.resolution_cache:
            return self.resolution_cache[cache_key].copy()
        
        # 执行拓扑排序
        result = self._topological_sort(plugin_names)
        
        # 缓存结果
        self.resolution_cache[cache_key] = result.copy()
        
        return result
    
    def _generate_cache_key(self, plugin_names: List[str]) -> str:
        """生成缓存键"""
        # 包含版本信息确保缓存一致性
        data = {
            'plugins': sorted(plugin_names),
            'version': self.cache_version
        }
        return hashlib.md5(json.dumps(data, sort_keys=True).encode()).hexdigest()
    
    def _topological_sort(self, plugin_names: List[str]) -> List[str]:
        """优化的拓扑排序"""
        # 构建子图
        subgraph = {}
        in_degree = {}
        
        # 计算子图中每个节点的入度
        for plugin in plugin_names:
            in_degree[plugin] = 0
            subgraph[plugin] = set()
        
        for plugin in plugin_names:
            for dep in self.dependency_graph.get(plugin, set()):
                if dep in plugin_names:
                    subgraph[dep].add(plugin)
                    in_degree[plugin] += 1
        
        # Kahn算法
        queue = [plugin for plugin in plugin_names if in_degree[plugin] == 0]
        result = []
        
        while queue:
            current = queue.pop(0)
            result.append(current)
            
            for dependent in subgraph[current]:
                in_degree[dependent] -= 1
                if in_degree[dependent] == 0:
                    queue.append(dependent)
        
        # 检查循环依赖
        if len(result) != len(plugin_names):
            remaining = [p for p in plugin_names if p not in result]
            raise ValueError(f"检测到循环依赖: {remaining}")
        
        return result
    
    def _invalidate_cache(self):
        """清除缓存"""
        self.resolution_cache.clear()
        self.cache_version += 1
    
    def get_dependents(self, plugin_name: str) -> Set[str]:
        """获取依赖于指定插件的所有插件"""
        return self.reverse_graph.get(plugin_name, set()).copy()
    
    def has_circular_dependency(self, plugin_names: List[str]) -> bool:
        """检查是否存在循环依赖"""
        try:
            self._topological_sort(plugin_names)
            return False
        except ValueError:
            return True
    
    def get_dependency_tree(self, plugin_name: str) -> Dict:
        """获取依赖树"""
        def build_tree(name, visited):
            if name in visited:
                return {"name": name, "circular": True}
            
            visited.add(name)
            
            tree = {"name": name, "dependencies": []}
            for dep in self.dependency_graph.get(name, set()):
                tree["dependencies"].append(build_tree(dep, visited.copy()))
            
            return tree
        
        return build_tree(plugin_name, set())
```

```python
from typing import Dict, Any, Callable, Optional, Type, TypeVar
from abc import ABC, abstractmethod
import inspect
import asyncio

T = TypeVar('T')

class DIContainer:
    """依赖注入容器"""
    
    def __init__(self):
        self._services: Dict[str, Any] = {}
        self._singletons: Dict[str, Any] = {}
        self._factories: Dict[str, Callable] = {}
        self._interfaces: Dict[Type, str] = {}
        self._lifecycle_callbacks: Dict[str, Dict[str, Callable]] = {}
    
    def register_singleton(self, interface: Type[T], implementation: T, name: Optional[str] = None):
        """注册单例服务"""
        service_name = name or interface.__name__
        self._singletons[service_name] = implementation
        self._interfaces[interface] = service_name
        
        # 注册生命周期回调
        if hasattr(implementation, 'initialize'):
            self._add_lifecycle_callback(service_name, 'initialize', implementation.initialize)
        if hasattr(implementation, 'cleanup'):
            self._add_lifecycle_callback(service_name, 'cleanup', implementation.cleanup)
    
    def register_factory(self, interface: Type[T], factory_func: Callable[[], T], name: Optional[str] = None):
        """注册工厂函数"""
        service_name = name or interface.__name__
        self._factories[service_name] = factory_func
        self._interfaces[interface] = service_name
    
    def register_transient(self, interface: Type[T], implementation_class: Type[T], name: Optional[str] = None):
        """注册瞬态服务"""
        service_name = name or interface.__name__
        self._services[service_name] = implementation_class
        self._interfaces[interface] = service_name
    
    def get(self, interface: Type[T]) -> T:
        """获取服务实例"""
        service_name = self._interfaces.get(interface)
        if not service_name:
            raise ValueError(f"服务 {interface.__name__} 未注册")
        
        return self._resolve_service(service_name)
    
    def get_by_name(self, service_name: str) -> Any:
        """按名称获取服务"""
        return self._resolve_service(service_name)
    
    def _resolve_service(self, service_name: str) -> Any:
        """解析服务实例"""
        # 检查单例
        if service_name in self._singletons:
            return self._singletons[service_name]
        
        # 检查工厂
        if service_name in self._factories:
            return self._factories[service_name]()
        
        # 检查瞬态服务
        if service_name in self._services:
            service_class = self._services[service_name]
            return self._create_instance(service_class)
        
        raise ValueError(f"服务 {service_name} 未找到")
    
    def _create_instance(self, service_class: Type) -> Any:
        """创建服务实例（支持依赖注入）"""
        # 检查构造函数参数
        init_signature = inspect.signature(service_class.__init__)
        init_params = {}
        
        for param_name, param in init_signature.parameters.items():
            if param_name == 'self':
                continue
            
            # 尝试从容器中解析依赖
            param_type = param.annotation
            if param_type != inspect.Parameter.empty:
                try:
                    init_params[param_name] = self.get(param_type)
                except ValueError:
                    if param.default == inspect.Parameter.empty:
                        raise ValueError(f"无法解析服务 {service_class.__name__} 的依赖 {param_name}")
        
        return service_class(**init_params)
    
    def _add_lifecycle_callback(self, service_name: str, phase: str, callback: Callable):
        """添加生命周期回调"""
        if service_name not in self._lifecycle_callbacks:
            self._lifecycle_callbacks[service_name] = {}
        self._lifecycle_callbacks[service_name][phase] = callback
    
    async def initialize_all(self):
        """初始化所有服务"""
        for service_name, callbacks in self._lifecycle_callbacks.items():
            if 'initialize' in callbacks:
                callback = callbacks['initialize']
                try:
                    if asyncio.iscoroutinefunction(callback):
                        await callback()
                    else:
                        callback()
                except Exception as e:
                    logger.error(f"服务 {service_name} 初始化失败: {e}")
    
    async def cleanup_all(self):
        """清理所有服务"""
        for service_name, callbacks in self._lifecycle_callbacks.items():
            if 'cleanup' in callbacks:
                callback = callbacks['cleanup']
                try:
                    if asyncio.iscoroutinefunction(callback):
                        await callback()
                    else:
                        callback()
                except Exception as e:
                    logger.error(f"服务 {service_name} 清理失败: {e}")
    
    def get_service_info(self) -> Dict[str, Any]:
        """获取服务信息"""
        return {
            'singletons': list(self._singletons.keys()),
            'factories': list(self._factories.keys()),
            'transients': list(self._services.keys()),
            'interfaces': {interface.__name__: name for interface, name in self._interfaces.items()}
        }
```

```python
from typing import Dict, Any, Optional, Type, TypeVar

class ConfigurationManager:
    """统一配置管理器"""

        """加载配置文件"""
        if config_file is None:
            config_file = f"{config_name}.yaml"
        
        config_path = self.config_dir / config_file
        
        if not config_path.exists():
            raise FileNotFoundError(f"配置文件不存在: {config_path}")
        
        # 读取配置文件
        with open(config_path, 'r', encoding='utf-8') as f:
            if config_path.suffix.lower() == '.yaml':
                config_data = yaml.safe_load(f)
            elif config_path.suffix.lower() == '.json':
                config_data = json.load(f)
            else:
                raise ValueError(f"不支持的配置文件格式: {config_path.suffix}")
        
        # 验证配置
        if config_name in self.validators:
            validator = self.validators[config_name]
            validated_config = validator(**config_data)
            self.configs[config_name] = validated_config
            return validated_config
        else:
            self.configs[config_name] = config_data
            return config_data
    
    def get_config(self, config_name: str, config_type: Type[T] = None) -> T:
        """获取配置"""
        if config_name not in self.configs:
            self.load_config(config_name)
        
        config = self.configs[config_name]
        
        if config_type and not isinstance(config, config_type):
            # 尝试转换类型
            if isinstance(config, dict):
                config = config_type(**config)
                self.configs[config_name] = config
            else:
                raise TypeError(f"配置 {config_name} 类型不匹配")
        
        return config
    
    def update_config(self, config_name: str, config_data: Dict[str, Any]):
        """更新配置"""
        if config_name in self.validators:
            validator = self.validators[config_name]
            validated_config = validator(**config_data)
            self.configs[config_name] = validated_config
        else:
            self.configs[config_name] = config_data
        
        # 触发变更监听器
        if config_name in self.watchers:
            self.watchers[config_name](config_data)
    
    def watch_config(self, config_name: str, callback: Callable[[Any], None]):
        """监听配置变更"""
        self.watchers[config_name] = callback
    
    def reload_config(self, config_name: str):
        """重新加载配置"""
        if config_name in self.config_cache:
            del self.config_cache[config_name]
        return self.load_config(config_name)
    
    def save_config(self, config_name: str, config_file: Optional[str] = None):
        """保存配置到文件"""
        if config_name not in self.configs:
            raise ValueError(f"配置 {config_name} 不存在")
        
        if config_file is None:
            config_file = f"{config_name}.yaml"
        
        config_path = self.config_dir / config_file
        config_data = self.configs[config_name]
        
        # 如果是 Pydantic 模型，转换为字典
        if isinstance(config_data, BaseModel):
            config_data = config_data.dict()
        
        with open(config_path, 'w', encoding='utf-8') as f:
            if config_path.suffix.lower() == '.yaml':
                yaml.safe_dump(config_data, f, default_flow_style=False, allow_unicode=True)
            elif config_path.suffix.lower() == '.json':
                json.dump(config_data, f, indent=2, ensure_ascii=False)
            else:
                raise ValueError(f"不支持的配置文件格式: {config_path.suffix}")
    
    def get_all_configs(self) -> Dict[str, Any]:
        """获取所有配置"""
        return self.configs.copy()
    
    def validate_all_configs(self) -> Dict[str, bool]:
        """验证所有配置"""
        results = {}
        for config_name, config_data in self.configs.items():
            try:
                if config_name in self.validators:
                    validator = self.validators[config_name]
                    validator(**config_data if isinstance(config_data, dict) else config_data.dict())
                results[config_name] = True
            except Exception as e:
                logger.error(f"配置 {config_name} 验证失败: {e}")
                results[config_name] = False
        return results
```

```python
from contextlib import asynccontextmanager

class OptimizedApplicationFactory:
    """优化的应用工厂"""
    
    def __init__(self, container: DIContainer, config_manager: ConfigurationManager):

        )

        # 配置路由
        self._configure_routes(app)

        # 注册应用到容器

        """创建生命周期上下文"""
        try:
            # 启动阶段
            await self._startup()
            yield
        finally:
            # 关闭阶段
            await self._shutdown()
    
    async def _startup(self):

        logger.info("应用启动中...")
        
        # 初始化容器中的所有服务
        await self.container.initialize_all()
        
        # 执行启动回调
        for callback in self.startup_callbacks:
            try:
                if asyncio.iscoroutinefunction(callback):
                    await callback()
                else:
                    callback()
            except Exception as e:
                logger.error(f"启动回调执行失败: {e}")
        
        logger.info("应用启动完成")
    
    async def _shutdown(self):
        """应用关闭"""
        logger.info("应用关闭中...")
        
        # 执行关闭回调
        for callback in self.shutdown_callbacks:
            try:
                if asyncio.iscoroutinefunction(callback):
                    await callback()
                else:
                    callback()
            except Exception as e:
                logger.error(f"关闭回调执行失败: {e}")
        
        # 清理容器中的所有服务
        await self.container.cleanup_all()
        
        logger.info("应用关闭完成")
    
    def _configure_middleware(self, app: FastAPI, environment: str):
        """配置中间件"""
        middleware_config = self.config_manager.get_config(f"middleware_{environment}")
        
        # 按顺序添加中间件
        for middleware_name in middleware_config.order:
            middleware_class = middleware_config.middleware[middleware_name]
            middleware_settings = middleware_config.settings.get(middleware_name, {})
            
            # 从容器获取中间件实例
            middleware_instance = self.container.get(middleware_class)
            app.add_middleware(middleware_class, **middleware_settings)
    
    def _configure_routes(self, app: FastAPI):

        # 从容器获取路由配置
        route_config = self.config_manager.get_config("routes")
        
        for route_group in route_config.groups:
            # 从容器获取控制器
            controller = self.container.get(route_group.controller)
            
            # 注册路由
            app.include_router(
                controller.router,
                prefix=route_group.prefix,
                tags=route_group.tags
            )
    
    def _configure_exception_handlers(self, app: FastAPI):
        """配置异常处理"""
        exception_config = self.config_manager.get_config("exceptions")
        
        for exception_type, handler_name in exception_config.handlers.items():
            handler = self.container.get_by_name(handler_name)
            app.add_exception_handler(exception_type, handler)
    
    def add_startup_callback(self, callback: Callable):
        """添加启动回调"""
        self.startup_callbacks.append(callback)
    
    def add_shutdown_callback(self, callback: Callable):
        """添加关闭回调"""
        self.shutdown_callbacks.append(callback)
    
    def add_middleware(self, middleware_class: type, **kwargs):
        """添加中间件"""
        self.middleware_stack.append((middleware_class, kwargs))
```

```python
import asyncio
import time
import hashlib
from typing import Dict, Optional, Any
from collections import deque
import threading

class HighPerformanceRequestHandler:
    """高性能请求处理器"""
    
    def __init__(self, pool_size: int = 1000):
        self.pool_size = pool_size
        self.request_id_pool = deque()
        self.pool_lock = threading.Lock()
        self.counter = 0
        self.last_timestamp = 0
        
        # 预生成请求ID池
        self._populate_request_id_pool()
    
    def _populate_request_id_pool(self):
        """预生成请求ID池"""
        current_time = int(time.time() * 1000)  # 毫秒时间戳
        
        for i in range(self.pool_size):
            # 使用时间戳 + 计数器 + 随机数生成ID
            request_id = f"{current_time:x}{i:04x}{hash(threading.current_thread()) & 0xffff:04x}"
            self.request_id_pool.append(request_id)
    
    def get_request_id(self) -> str:
        """高性能请求ID生成"""
        with self.pool_lock:
            if self.request_id_pool:
                return self.request_id_pool.popleft()
        
        # 池空时使用计数器生成
        current_time = int(time.time() * 1000)
        if current_time != self.last_timestamp:
            self.counter = 0
            self.last_timestamp = current_time
        else:
            self.counter += 1
        
        return f"{current_time:x}{self.counter:04x}{threading.get_ident() & 0xffff:04x}"
    
    def return_request_id(self, request_id: str):
        """回收请求ID（可选）"""
        with self.pool_lock:
            if len(self.request_id_pool) < self.pool_size:
                self.request_id_pool.append(request_id)
    
    def refill_pool(self):
        """补充ID池"""
        with self.pool_lock:
            needed = self.pool_size - len(self.request_id_pool)
            if needed > 0:
                current_time = int(time.time() * 1000)
                for i in range(needed):
                    request_id = f"{current_time:x}{i:04x}{hash(threading.current_thread()) & 0xffff:04x}"
                    self.request_id_pool.append(request_id)
    
    def get_pool_stats(self) -> Dict[str, Any]:
        """获取池统计信息"""
        with self.pool_lock:
            return {
                "pool_size": self.pool_size,
                "available_ids": len(self.request_id_pool),
                "utilization": (self.pool_size - len(self.request_id_pool)) / self.pool_size,
                "counter": self.counter,
                "last_timestamp": self.last_timestamp
            }
```

#### 4.3.2 异步日志记录系统

```python
import asyncio
import logging
from typing import Dict, Any, Optional
from datetime import datetime
import json
from collections import deque

class AsyncLoggerManager:
    """异步日志管理器"""
    
    def __init__(self, 
                 batch_size: int = 100, 
                 flush_interval: float = 1.0,
                 max_queue_size: int = 10000):
        self.batch_size = batch_size
        self.flush_interval = flush_interval
        self.max_queue_size = max_queue_size
        
        self.log_queue = asyncio.Queue(maxsize=max_queue_size)
        self.batch_buffer = deque()
        self.worker_task = None
        self.stats = {
            'total_logs': 0,
            'dropped_logs': 0,
            'batches_processed': 0,
            'errors': 0
        }
        
        # 创建专用日志器
        self.logger = logging.getLogger('async_logger')
        self.error_logger = logging.getLogger('async_logger_errors')
    
    async def start(self):
        """启动异步日志工作器"""
        if self.worker_task is None:
            self.worker_task = asyncio.create_task(self._log_worker())
    
    async def stop(self):
        """停止异步日志工作器"""
        if self.worker_task:
            self.worker_task.cancel()
            try:
                await self.worker_task
            except asyncio.CancelledError:
                pass
            
            # 处理剩余日志
            await self._flush_remaining_logs()
    
    async def log_async(self, 
                       level: int, 
                       message: str, 
                       extra: Optional[Dict[str, Any]] = None,
                       request_id: Optional[str] = None):
        """异步日志记录"""
        log_entry = {
            'level': level,
            'message': message,
            'timestamp': datetime.now().isoformat(),
            'request_id': request_id,
            'extra': extra or {}
        }
        
        try:
            self.log_queue.put_nowait(log_entry)
            self.stats['total_logs'] += 1
        except asyncio.QueueFull:
            self.stats['dropped_logs'] += 1
            # 队列满时，丢弃最旧的日志
            try:
                self.log_queue.get_nowait()
                self.log_queue.put_nowait(log_entry)
            except asyncio.QueueEmpty:
                pass
    
    async def _log_worker(self):
        """日志处理工作器"""
        last_flush = asyncio.get_event_loop().time()
        
        while True:
            try:
                # 等待日志或超时
                timeout = max(0.1, self.flush_interval - (asyncio.get_event_loop().time() - last_flush))
                log_entry = await asyncio.wait_for(self.log_queue.get(), timeout=timeout)
                
                self.batch_buffer.append(log_entry)
                
                # 检查是否需要刷新
                current_time = asyncio.get_event_loop().time()
                should_flush = (
                    len(self.batch_buffer) >= self.batch_size or
                    current_time - last_flush >= self.flush_interval
                )
                
                if should_flush:
                    await self._flush_batch()
                    last_flush = current_time
                    
            except asyncio.TimeoutError:
                # 超时刷新
                if self.batch_buffer:
                    await self._flush_batch()
                    last_flush = asyncio.get_event_loop().time()
                    
            except Exception as e:
                self.stats['errors'] += 1
                self.error_logger.error(f"日志工作器错误: {e}")
    
    async def _flush_batch(self):
        """刷新批次日志"""
        if not self.batch_buffer:
            return
        
        batch = list(self.batch_buffer)
        self.batch_buffer.clear()
        
        try:
            # 批量写入日志
            for log_entry in batch:
                self.logger.log(
                    log_entry['level'],
                    log_entry['message'],
                    extra={
                        'request_id': log_entry.get('request_id'),
                        'timestamp': log_entry['timestamp'],
                        **log_entry.get('extra', {})
                    }
                )
            
            self.stats['batches_processed'] += 1
            
        except Exception as e:
            self.stats['errors'] += 1
            self.error_logger.error(f"批量日志写入失败: {e}")
    
    async def _flush_remaining_logs(self):
        """处理剩余日志"""
        while not self.log_queue.empty():
            try:
                log_entry = self.log_queue.get_nowait()
                self.batch_buffer.append(log_entry)
            except asyncio.QueueEmpty:
                break
        
        if self.batch_buffer:
            await self._flush_batch()
    
    def get_stats(self) -> Dict[str, Any]:
        """获取统计信息"""
        return {
            **self.stats,
            'queue_size': self.log_queue.qsize(),
            'buffer_size': len(self.batch_buffer),
            'drop_rate': self.stats['dropped_logs'] / max(1, self.stats['total_logs']) * 100
```

```python
class OptimizedBaseController:

            'cache_misses': 0
        }
    
    async def handle_request(self, 
                           request: Request,
                           handler_func: callable,
                           *args, 
                           **kwargs) -> Dict[str, Any]:
        """优化的请求处理"""
        start_time = time.time()
        request_id = self.request_handler.get_request_id()
        
        # 设置请求上下文
        request.state.request_id = request_id
        request.state.start_time = start_time
        
        try:
            # 检查缓存
            cache_key = self._get_cache_key(request, handler_func, args, kwargs)
            cached_response = self._get_cached_response(cache_key)
            
            if cached_response is not None:
                self.stats['cache_hits'] += 1
                await self._log_request_async(request, "cache_hit", response_time=time.time() - start_time)
                return cached_response
            
            self.stats['cache_misses'] += 1
            
            # 执行处理逻辑
            if asyncio.iscoroutinefunction(handler_func):
                result = await handler_func(request, *args, **kwargs)
            else:
                result = handler_func(request, *args, **kwargs)
            
            # 构建响应
            response = self._build_success_response(result)
            
            # 缓存响应
            self._cache_response(cache_key, response)
            
            # 记录成功
            self.stats['successful_requests'] += 1
            self.stats['total_requests'] += 1
            
            # 异步日志记录
            response_time = time.time() - start_time
            await self._log_request_async(request, "success", response_time=response_time)
            
            # 更新平均响应时间
            self._update_avg_response_time(response_time)
            
            return response
            
        except HTTPException as e:
            # HTTP异常处理
            self.stats['failed_requests'] += 1
            self.stats['total_requests'] += 1
            
            response_time = time.time() - start_time
            await self._log_request_async(request, "http_error", 
                                        error_code=e.status_code,
                                        error_message=str(e.detail),
                                        response_time=response_time)
            
            return self._build_error_response(e.status_code, str(e.detail))
            
        except Exception as e:
            # 通用异常处理
            self.stats['failed_requests'] += 1
            self.stats['total_requests'] += 1
            
            response_time = time.time() - start_time
            await self._log_request_async(request, "error", 
                                        error_type=type(e).__name__,
                                        error_message=str(e),
                                        response_time=response_time)
            
            return self._build_error_response(500, "内部服务器错误")
            
        finally:
            # 回收请求ID
            self.request_handler.return_request_id(request_id)
    
    def _get_cache_key(self, request: Request, handler_func: callable, args: tuple, kwargs: dict) -> str:
        """生成缓存键"""
        # 简化版本，实际应用中需要更复杂的逻辑
        method = request.method
        url = str(request.url)
        func_name = handler_func.__name__
        
        # 对于GET请求，包含查询参数
        if method == "GET":
            query_params = dict(request.query_params)
            cache_key = f"{func_name}:{method}:{url}:{hash(frozenset(query_params.items()))}"
        else:
            cache_key = f"{func_name}:{method}:{url}"
        
        return cache_key
    
    def _get_cached_response(self, cache_key: str) -> Optional[Dict[str, Any]]:
        """获取缓存响应"""
        cached_item = self.response_cache.get(cache_key)
        if cached_item:
            response, timestamp = cached_item
            # 检查是否过期（5分钟）
            if time.time() - timestamp < 300:
                return response
            else:
                del self.response_cache[cache_key]
        return None
    
    def _cache_response(self, cache_key: str, response: Dict[str, Any]):
        """缓存响应"""
        # 限制缓存大小
        if len(self.response_cache) > 1000:
            # 移除最旧的缓存项
            oldest_key = min(self.response_cache.keys(), 
                           key=lambda k: self.response_cache[k][1])
            del self.response_cache[oldest_key]
        
        self.response_cache[cache_key] = (response, time.time())
    
    async def _log_request_async(self, request: Request, action: str, **kwargs):
        """异步日志记录"""
        log_data = {
            'method': request.method,
            'url': str(request.url),
            'user_agent': request.headers.get('user-agent', ''),
            'client_ip': request.client.host if request.client else 'unknown',
            'action': action,
            **kwargs
        }
        
        await self.async_logger.log_async(
            level=logging.INFO,
            message=f"请求处理: {action}",
            extra=log_data,
            request_id=request.state.request_id
        )
    
    def _build_success_response(self, data: Any) -> Dict[str, Any]:
        """构建成功响应"""
        return {
            "success": True,
            "message": "操作成功",
            "data": data,
            "timestamp": datetime.now().isoformat()
        }
    
    def _build_error_response(self, status_code: int, message: str) -> Dict[str, Any]:
        """构建错误响应"""
        return {
            "success": False,
            "message": message,
            "error_code": status_code,
            "timestamp": datetime.now().isoformat()
        }
    
    def _update_avg_response_time(self, response_time: float):
        """更新平均响应时间"""
        if self.stats['total_requests'] > 0:
            current_avg = self.stats['avg_response_time']
            total_requests = self.stats['total_requests']
            
            # 计算新的平均值
            new_avg = (current_avg * (total_requests - 1) + response_time) / total_requests
            self.stats['avg_response_time'] = new_avg
    
    def get_performance_stats(self) -> Dict[str, Any]:
        """获取性能统计"""
        return {
            **self.stats,
            'success_rate': self.stats['successful_requests'] / max(1, self.stats['total_requests']) * 100,
            'error_rate': self.stats['failed_requests'] / max(1, self.stats['total_requests']) * 100,
            'cache_hit_rate': self.stats['cache_hits'] / max(1, self.stats['cache_hits'] + self.stats['cache_misses']) * 100,
            'request_handler_stats': self.request_handler.get_pool_stats(),
            'logger_stats': self.async_logger.get_stats()
        }
    
    def clear_cache(self):
        """清除缓存"""
        self.response_cache.clear()
        self.stats['cache_hits'] = 0
        self.stats['cache_misses'] = 0
```

```python
from typing import Optional, List, Dict
import threading
import time
from collections import deque
from concurrent.futures import ThreadPoolExecutor

class LockFreeMemoryPool:
    """无锁内存池实现"""
    
    def __init__(self, pool_size: int = 1000, segment_count: int = 4):
        self.pool_size = pool_size
        self.segment_count = segment_count
        self.segment_size = pool_size // segment_count
        
        # 创建分段池
        self.segments = []
        for i in range(segment_count):
            segment = {
                'free_buffers': deque(),
                'used_buffers': set(),
                'lock': threading.Lock(),
                'stats': {'allocations': 0, 'deallocations': 0}
            }
            
            # 预分配缓冲区
            for j in range(self.segment_size):
                buffer = FrameBuffer(buffer_id=f"seg{i}_buf{j}")
                segment['free_buffers'].append(buffer)
            
            self.segments.append(segment)
        
        self.allocation_strategy = "round_robin"
        self.current_segment = 0
        self.stats = {
            'total_allocations': 0,
            'total_deallocations': 0,
            'allocation_failures': 0,
            'segment_utilization': [0] * segment_count
        }
    
    def get_buffer(self, frame_id: int, timestamp: float, stream_id: str = "") -> Optional['FrameBuffer']:
        """获取缓冲区"""
        if self.allocation_strategy == "round_robin":
            return self._get_buffer_round_robin(frame_id, timestamp, stream_id)
        elif self.allocation_strategy == "least_used":
            return self._get_buffer_least_used(frame_id, timestamp, stream_id)
        else:
            return self._get_buffer_random(frame_id, timestamp, stream_id)
    
    def _get_buffer_round_robin(self, frame_id: int, timestamp: float, stream_id: str) -> Optional['FrameBuffer']:
        """轮询分配策略"""
        start_segment = self.current_segment
        
        for i in range(self.segment_count):
            segment_idx = (start_segment + i) % self.segment_count
            segment = self.segments[segment_idx]
            
            # 尝试从当前段获取缓冲区
            with segment['lock']:
                if segment['free_buffers']:
                    buffer = segment['free_buffers'].popleft()
                    segment['used_buffers'].add(buffer)
                    segment['stats']['allocations'] += 1
                    
                    # 更新下一个分配段
                    self.current_segment = (segment_idx + 1) % self.segment_count
                    
                    # 初始化缓冲区
                    buffer.initialize(frame_id, timestamp, stream_id)
                    
                    self.stats['total_allocations'] += 1
                    return buffer
        
        # 所有段都没有可用缓冲区
        self.stats['allocation_failures'] += 1
        return None
    
    def _get_buffer_least_used(self, frame_id: int, timestamp: float, stream_id: str) -> Optional['FrameBuffer']:
        """最少使用分配策略"""
        best_segment = None
        best_count = float('inf')
        
        for segment in self.segments:
            with segment['lock']:
                used_count = len(segment['used_buffers'])
                if used_count < best_count and segment['free_buffers']:
                    best_count = used_count
                    best_segment = segment
        
        if best_segment:
            with best_segment['lock']:
                if best_segment['free_buffers']:
                    buffer = best_segment['free_buffers'].popleft()
                    best_segment['used_buffers'].add(buffer)
                    best_segment['stats']['allocations'] += 1
                    
                    buffer.initialize(frame_id, timestamp, stream_id)
                    self.stats['total_allocations'] += 1
                    return buffer
        
        self.stats['allocation_failures'] += 1
        return None
    
    def return_buffer(self, buffer: 'FrameBuffer'):
        """归还缓冲区"""
        segment_idx = self._get_buffer_segment(buffer)
        if segment_idx is not None:
            segment = self.segments[segment_idx]
            
            with segment['lock']:
                if buffer in segment['used_buffers']:
                    segment['used_buffers'].remove(buffer)
                    buffer.cleanup()
                    segment['free_buffers'].append(buffer)
                    segment['stats']['deallocations'] += 1
                    self.stats['total_deallocations'] += 1
    
    def _get_buffer_segment(self, buffer: 'FrameBuffer') -> Optional[int]:
        """获取缓冲区所属段"""
        buffer_id = buffer.buffer_id
        if buffer_id.startswith('seg'):
            try:
                segment_idx = int(buffer_id.split('_')[0][3:])
                return segment_idx
            except (ValueError, IndexError):
                pass
        return None
    
    def get_pool_stats(self) -> Dict:
        """获取池统计信息"""
        segment_stats = []
        for i, segment in enumerate(self.segments):
            with segment['lock']:
                segment_stats.append({
                    'segment_id': i,
                    'free_buffers': len(segment['free_buffers']),
                    'used_buffers': len(segment['used_buffers']),
                    'utilization': len(segment['used_buffers']) / self.segment_size * 100,
                    'allocations': segment['stats']['allocations'],
                    'deallocations': segment['stats']['deallocations']
                })
        
        return {
            'total_stats': self.stats,
            'segment_stats': segment_stats,
            'allocation_strategy': self.allocation_strategy,
            'total_utilization': sum(s['utilization'] for s in segment_stats) / len(segment_stats)
        }
    
    def optimize_allocation_strategy(self):
        """优化分配策略"""
        stats = self.get_pool_stats()
        
        # 根据统计信息调整策略
        avg_utilization = stats['total_utilization']
        utilization_variance = sum((s['utilization'] - avg_utilization) ** 2 for s in stats['segment_stats']) / len(stats['segment_stats'])
        
        if utilization_variance > 100:  # 利用率差异较大
            self.allocation_strategy = "least_used"
        elif avg_utilization > 80:  # 高利用率
            self.allocation_strategy = "round_robin"
        else:
            self.allocation_strategy = "round_robin"
    
    def cleanup_expired_buffers(self, max_age_seconds: float = 300):
        """清理过期缓冲区"""
        current_time = time.time()
        cleaned_count = 0
        
        for segment in self.segments:
            with segment['lock']:
                expired_buffers = []
                for buffer in segment['used_buffers']:
                    if current_time - buffer.timestamp > max_age_seconds:
                        expired_buffers.append(buffer)
                
                for buffer in expired_buffers:
                    segment['used_buffers'].remove(buffer)
                    buffer.cleanup()
                    segment['free_buffers'].append(buffer)
                    cleaned_count += 1
        
        return cleaned_count
```
